\begin{tikzpicture}[font=\scriptsize, x=1mm, y=1mm]

\node[circle,draw=cNeut,thick,fill=white,inner sep=1.2pt] (sum) at (3,0) {$\Sigma$};

  {\textbf{DRIVE}\\\textbf{AMP}};

  {\textbf{KLYSTRON}\\{\tiny \qty{1.5}{\mega\watt}}};

  {\textbf{COMBINING}\\\textbf{NETWORK}\\{\tiny weighted vector sum}};
